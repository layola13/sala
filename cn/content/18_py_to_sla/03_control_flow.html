布尔、条件与循环

布尔、逻辑运算、if/elif/else 与三元形式

psa 的逻辑运算符写作 and/or/not；SLA 在 bool 值上用 &&/||/!。普查口径：psa 的 and/or/not 已支持（纯布尔时降为 SLA 运算符）。Python 的 elif 就是 SLA 的 else if，块用花括号而非缩进。Python 的三元式 x if c else y 对应 SLA 的 if 表达式：if c { x } else { y }，可放在任何需要值的位置。

psaSLA
def grade(score):
    if score >= 90:
        return 1
    elif score >= 60:
        return 2
    else:
        return 3

a = True
b = False
print(a and not b)            # True
print(a or b)                 # True
print(1 if 95 >= 60 else 0)  # 1
print(grade(95))              # 1
print(grade(70))              # 2
print(grade(10))              # 3
 
@import "sa_std/io/print.sai"
@import "sa_std/fmt.sai"

fn grade(score: i32) -> i32 {
    if score >= 90 {
        return 1;
    } else if score >= 60 {
        return 2;
    } else {
        return 3;
    }
}

fn main() -> i32 {
    let a = true;
    let b = false;
    println("{}", a && !b);
    println("{}", a || b);
    println("{}", if 95 >= 60 { 1 } else { 0 });
    println("{}", grade(95));
    println("{}", grade(70));
    println("{}", grade(10));
    return 0;
}

实测输出（SLA）：true、true、1、1、2、3，退出码 0。

while、break 与 continue

循环形式一一对应。普查口径：psa 的 while 带 break/continue 已支持；while/else 与 for/else 子句是 psa 已登记的前端缺口（响亮拒绝，低频）。SLA 同样没有循环 else。

psaSLA
total = 0
i = 0
while i < 10:
    i = i + 1
    if i == 3:
        continue
    if i == 8:
        break
    total = total + i
print(total)  # 25
 
@import "sa_std/io/print.sai"
@import "sa_std/fmt.sai"

fn main() -> i32 {
    let sum = 0;
    let i = 0;
    while i < 10 {
        i = i + 1;
        if i == 3 { continue; };
        if i == 8 { break; };
        sum = sum + i;
    }
    println("{}", sum);
    return 0;
}

实测输出（SLA）：25，退出码 0。

for 与 range

psa 的 for i in range(...) 对应 SLA 的 for i in a..b——区间是循环语法的一部分，没有独立的 range 值。psa 侧普查状态为部分支持：range 一参/二参在 for 头（及推导式/构造器）里可用，遍历 list 变量与字符串可用，但 range 对象作一般值仍是缺口。按下标走集合时两边都用长度：psa range(len(a))，SLA 0..len(a)。

psaSLA
total = 0
for i in range(1, 5):
    total = total + i
print(total)  # 10

a = [10, 20, 30]
total = 0
for i in range(len(a)):
    total = total + a[i]
print(total)  # 60
 
@import "sa_std/io/print.sai"
@import "sa_std/fmt.sai"

fn main() -> i32 {
    let sum = 0;
    for i in 1..5 {
        sum = sum + i;
    }
    println("{}", sum);
    let a = [10, 20, 30];
    let total = 0;
    for i in 0..len(a) {
        total = total + a[i];
    }
    println("{}", total);
    return 0;
}

实测输出（SLA）：10、60，退出码 0。

本页要点

  • and/or/not ↔ &&/||/!；elif ↔ else if；三元式 ↔ if 表达式。
  • while/break/continue 直接对应；循环 else 两边都没有（psa 侧为登记缺口）。
  • for i in range(a, b) ↔ for i in a..b；psa 的 range 作独立值是缺口（普查：部分）。
