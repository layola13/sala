常见配方：从打印到 HTTP 的完整示例

这一页把最常用的七件事各给一段可直接对照的代码：Sla 写法与 SA 汇编写法成对出现， 全部摘自仓库里的 demo / test 文件并注明出处路径。找不到真实调用处的配方会明确标注， 只给出基于契约签名的调用形式。

hello world 与 println 格式化

Sla 写法，摘自 demos/rosetta/01_hello_world/main.sla：

@import "sa_std/io/print.sai"
@import "sa_std/fmt.sai"
@import "sa_std/string.sa"

fn main() -> i32 {
    println("hello, saasm");
    return 0;
};

占位符格式化，摘自 demos/rosetta/05_struct/main.sla（04_loop/main.sla 里还有 println("[{},{},{},{}]", zeroed[0], ...) 的多参数版本）：

let point = create_point(10, 20);
println("({},{})", point.x, point.y);

SA 汇编写法——同一目录下编译器生成的 main.sa 展示了 println("{}", result) 的真身（摘自 demos/rosetta/02_mutability/main.sa）：

tmp_4 = call @sa_fmt_u64(result, 10)
tmp_5 = call @sa_fmt_buffer_data(tmp_4)
tmp_6 = call @sa_fmt_buffer_len(tmp_4)
tmp_7 = tmp_5
call @sa_print_bytes(&tmp_7, tmp_6)
call @sa_fmt_buffer_free(^tmp_4)
@const SLA_STR_0 = utf8:"\n"
call @sa_print_bytes(&SLA_STR_0, 1)

手写汇编时最小形式就是一条 call @sa_print_bytes(&msg, len)， 见 sci/tests/llvm2sa_expected_hello.sa 的 call @sa_print_bytes(&HELLO_MSG, 13)。

读文件到字符串

Sla 写法 A：经项目侧包装宏。包装宏定义摘自 tests/import_fixtures/pkgjson/fs_helpers.sa：

@import "sa_std/fs.sa"

[MACRO] SLA_FS_READ_TO_STRING %out_buffer, %path, %path_len, %max_bytes
    __sla_fs_read_status_%out_buffer = 0
    EXPAND FS_READ_TO_STRING __sla_fs_read_status_%out_buffer, %out_buffer, %path, %path_len, %max_bytes
    !__sla_fs_read_status_%out_buffer
[END_MACRO]

调用处摘自 tests/test_unit_imported_fs_read_direct.sla：

@import "sa_std/string.sa"
@import "sa_std/fs.sa"
@import "import_fixtures/pkgjson/fs_helpers.sa"

fn imported_fs_read_missing(path_ptr: ptr, path_len: int) -> int {
    let buf = SLA_FS_READ_TO_STRING(path_ptr, path_len, 16);
    let len = SLA_FS_BUFFER_LEN(buf);
    return len;
}

Sla 写法 B：直接调 extern 契约（fallible 句柄，返回 0 表示失败）， 摘自 tests/test_unit_fallible_extern_payload_direct.sla：

@import "sa_std/fs.sai"

fn fallible_extern_read_len(path_ptr: ptr, path_len: u64) -> u64 {
    let buffer = sa_fs_read_file(path_ptr, path_len, 4096);
    switch buffer {
        0 => {
            return 0;
        },
        default => {},
    };
    let data_len = sa_fs_read_buffer_len(buffer);
    let free_status = sa_fs_read_buffer_free(buffer);
    if free_status != 0 {
        return 0;
    };
    return data_len;
}

两种契约签名并存于 sci/sa_std/fs.sai： sa_std_fs_read_to_string(&path: ptr, path_len: u64, max_bytes: u64, &out_handle: ptr) -> i32 （状态 + 出参槽位，宏用这种）与 sa_fs_read_to_string(&path: ptr, path_len: u64, max_bytes: u64) -> u64! （fallible 句柄，Sla 直接调用用这种）。宏清单见 fs 一章。

环境变量读取 / 命令行参数

契约核心（摘自 sci/sa_std/env.sai）：

@extern sa_env_get(key: ptr, key_len: u64) -> u64
@extern sa_env_has(key: ptr, key_len: u64) -> i32
@extern sa_env_args_json() -> u64

SA 汇编写法（真实测试），摘自 sci/tests/unit_framework/std_env_macro_surface.sa，文件顶部为 @import "sa_std/env.sa"：

@test "sa_std env args JSON macro"():
L_ARGS_ENTRY:
    EXPAND ENV_ARGS_JSON args_buffer
    EXPAND ENV_BUFFER_DATA args_ptr, args_buffer
    EXPAND ENV_BUFFER_LEN args_len, args_buffer

    args_buffer_nonzero = ne args_buffer, 0
    args_ptr_nonzero = ne args_ptr, 0
    args_len_nonzero = ne args_len, 0
    args_shape_ready = and args_buffer_nonzero, args_ptr_nonzero
    args_ready = and args_shape_ready, args_len_nonzero

    br args_ready -> L_ARGS_CHECK_JSON, L_ARGS_FAIL_READY

L_ARGS_CHECK_JSON:
    json_shape_ok = ne args_len, 0
    EXPAND ENV_BUFFER_FREE args_free_status, args_buffer
    EXPAND ENV_STATUS_IS_OK args_free_ok, args_free_status
注意仓库中没有 .sla 调用 env 契约的示例。 基于上述签名的 Sla 调用形式是：let args = sa_env_args_json(); 之后依次取 sa_env_buffer_len(args) 与 sa_env_buffer_data(args)，最后 sa_env_buffer_free(args)；sa_env_get(key, key_len) 同理。 返回的是 JSON 数组文本缓冲，需自行解析（见下文 JSON 配方）。

子进程运行并捕获输出

契约核心（摘自 sci/sa_std/process.sai；注意前缀是 sa_std_process_*）：

@extern sa_std_process_run(&argv: ptr, argv_len: u64, &out_handle: ptr) -> i32
@extern sa_std_process_exec_capture(&argv: ptr, argv_len: u64, &out_code: ptr, &out_stdout: ptr, &out_stderr: ptr) -> i32
@extern sa_std_process_close(handle: u64) -> i32

SA 汇编写法（真实测试），摘自 sci/tests/unit_framework/std_process_macro_surface.sa，文件顶部为 @import "sa_std/process.sa"。argv 是 SaProcessArgv 记录数组， 用宏逐项填充：

    argv = stack_alloc TRUE_ARGV2_SIZE
    output = stack_alloc ProcessOutput_SIZE
    argv0 = stack_alloc Slice_SIZE
    store argv0+Slice_ptr, &TRUE_BIN as ptr
    store argv0+Slice_len, TRUE_BIN_LEN as u64
    EXPAND PROCESS_ARGV_SET_SLICE argv, 0, argv0
    EXPAND PROCESS_ARGV_SET_CONST argv, 1, TRUE_ARG, TRUE_ARG_LEN

    EXPAND PROCESS_EXEC_CAPTURE_OUTPUT capture_status, output, argv, 2
    EXPAND PROCESS_STATUS_IS_OK capture_ok, capture_status

L_CAPTURE_OK:
    EXPAND PROCESS_OUTPUT_STATUS_PTR output_status_ptr, output
    EXPAND PROCESS_EXIT_STATUS_CODE output_code, output_status_ptr
    EXPAND PROCESS_OUTPUT_STDOUT_BUFFER stdout_buffer, output
    EXPAND PROCESS_OUTPUT_STDOUT_DATA stdout_ptr, output
    EXPAND PROCESS_OUTPUT_STDOUT_LEN stdout_len, output
    EXPAND PROCESS_OUTPUT_FREE free_status, output

Sla 侧导入该契约本身有真实例子—— tests/test_unit_std_import.sla 里 let pid = sa_std_process_id();； 但 run / exec_capture 在仓库中暂无 .sla 调用示例， 其 Sla 调用形式需按上面的出参指针签名自行封装。详见 process 一章。

TCP 连接或 HTTP/2 请求

契约里有两套面：完整的 sa_std_net_* 与简化的 fallible 面 sa_net_*（摘自 sci/sa_std/net.sai）：

@extern sa_net_tcp_connect(&host: ptr, host_len: u64, port: u32) -> u64!
@extern sa_net_tcp_listener_bind(&host: ptr, host_len: u64, port: u16) -> u64!
@extern sa_net_tcp_listener_accept(listener: u64) -> u64!
@extern sa_net_tcp_stream_write_all(stream: u64, &buf: ptr, len: u64) -> i32!
@extern sa_net_tcp_stream_read_exact(stream: u64, &buf: ptr, len: u64) -> i32!
@extern sa_net_tcp_stream_close(stream: u64) -> i32!

SA 汇编写法（真实回环测试，节选，省略了状态检查与跳转），摘自 sci/tests/unit_framework/support/stdlib_surface.sa 的 @support_net_surface()。fallible 返回在这里表现为"状态字 + 句柄"两个槽位， 本例中 +0 为状态、+8 为句柄：

L_ENTRY:
    listener_res = call @sa_net_tcp_listener_bind(&NET_LOOPBACK, NET_LOOPBACK_LEN, 0)
    listener_status = load listener_res+0 as u32
    listener = load listener_res+8 as u64
    ...
L_ADDR:
    addr_res = call @sa_net_tcp_listener_local_addr(listener)
    ...
    port = call @sa_net_addr_port(addr)
    host_copy = alloc host_len
    ...
L_CONNECT:
    connect_res = call @sa_net_tcp_connect(&host_copy, host_len, port)
    connect_status = load connect_res+0 as u32
    client = load connect_res+8 as u64
    ...
L_ACCEPT:
    accept_res = call @sa_net_tcp_listener_accept(listener)
    accept_status = load accept_res+0 as u32
    server = load accept_res+8 as u64

仓库中没有 .sla 调用 net 契约的示例。HTTP/2 同样仓库中暂无调用示例， 只有契约与宏实现；调用形式按 sci/sa_std/http2.sai 签名为 sa_std_http2_client_request(&url: ptr, url_len: u64, &method: ptr, method_len: u64, &body: ptr, body_len: u64, &out_handle: ptr) -> i32，SA 侧对应宏是 sci/sa_std/http2.sa 的 [MACRO] HTTP2_CLIENT_REQUEST %out_status, %out_buffer, %url, %url_len, %method, %method_len, %body, %body_len，结果缓冲用 HTTP2_BUFFER_DATA / HTTP2_BUFFER_LEN / HTTP2_BUFFER_FREE 三件套取出。选项与错误面见 net 一章。

JSON 解析取值

Sla 写法 A：直接调契约函数。摘自 tests/test_unit_fallible_extern_payload_direct.sla：

@import "sa_std/fs.sai"
@import "sa_std/encoding/json.sai"
@import "sa_std/ptr.sa"
@import "sa_std/string.sa"

fn fallible_extern_json_free_status(node: ptr, owned: bool) -> i32 {
    if owned {
        return sa_json_free(node);
    };
    return 0;
}

@test "fallible consuming ptr extern closes a handle once"() {
    let root = sa_json_parse(STR_PTR("[{}]"), STR_LEN("[{}]") as u64);
    if PTR_IS_NONNULL(root) == false {
        panic(43013);
    };
    var child: ptr;
    child = PTR_NULL();
    let child_status = sa_json_array_get(root, 0, &child);
    if child_status != 0 || PTR_IS_NONNULL(child) == false {
        panic(43014);
    };
    let status = fallible_extern_json_free_status(child, true);
    if status != 0 {
        panic(43015);
    };
}

Sla 写法 B：经 JSON 宏。摘自 tests/test_unit_imported_json_direct.sla：

@import "sa_std/string.sa"
@import "sa_std/encoding/json.sa"
@import "import_fixtures/pkgjson/json_helpers.sa"

fn imported_json_get_type(data: ptr, data_len: int) -> int {
    let root = JSON_PARSE(data, data_len);
    if JSON_KIND(root) < 0 {
        return -1;
    };
    let t = SLA_JSON_OBJECT_GET(root, STR_PTR("types"), STR_LEN("types"));
    return JSON_KIND(t);
}

@test "imported json direct sab"() {
    let got = imported_json_get_type("{\"types\":\"x\"}", 13);
    if got != 5 { panic(39022); };
}

其中 SLA_JSON_OBJECT_GET 是项目侧包装宏，本质是把出参槽位包起来再调 sa_json_object_get(node, &key, key_len, &out_handle)，见 tests/import_fixtures/pkgjson/json_helpers.sa。按键取标量可走更短的 sa_json_object_get_i64 / _get_f64 / _get_string / _get_bool。 完整契约见 encoding 一章。

断言测试

契约全文（sci/sa_std/testing/assert.sai，共四个 extern）：

@extern sa_assert_eq_i64(actual: i64, expected: i64, code: i32) -> void
@extern sa_assert_eq_i64_at(actual: i64, expected: i64, code: i32, &file: ptr, file_len: u64, line: u32, col: u32) -> void
@extern sa_test_debug_i64(&name: ptr, name_len: u64, value: i64) -> void
@extern sa_test_fallible_i32_value(value: i32) -> i32!

SA 汇编写法（真实测试），摘自 sci/tests/unit_framework/std_fs_macro_surface.sa，文件顶部为 @import "sa_std/testing/assert.sai"；失败分支先用 sa_test_debug_i64 记录现场标量，再用断言收口：

L_FAIL_READ:
    call @sa_test_debug_i64(&FS_DBG_WRITE_STATUS, FS_DBG_WRITE_STATUS_LEN, write_file_status)
    call @sa_assert_eq_i64(write_file_status, SA_FS_OK, 940)
    call @sa_assert_eq_i64(exists_after_write, 1, 940)
    call @sa_assert_eq_i64(read_len, FS_TEST_HELLO_LEN, 940)
    call @sa_assert_eq_i64(read_eq, 1, 940)

带源码位置的版本用法见 sci/docs/unit_test_framework.md： call @sa_assert_eq_i64_at(raw_status, 1, 103, *THIS_FILE, 14, 8, 5)， runner 会把它解析成 panic location；失败时输出 PANIC[code]: expected=<expected> actual=<actual>。

Sla 写法——测试体直接写 @test 加条件 panic， 摘自 demos/rosetta/01_hello_world/main.sla：

@test "rosetta 001 01_hello_world"() {
    if str_eq("hello, saasm", "hello, saasm") != true {
        panic(1);
    };
};
注意sa test 用多进程隔离每个 @test，单个断言失败不会中断其余测试；断言细节与 testing 一章、 API 总索引可交叉查阅。
